## Use it

Import `SwiftCN` to use `CNDropdownMenu` directly. This complete example also runs in the native component gallery.

```swift title="DropdownMenuExample.swift"
import SwiftUI
import SwiftCN

struct DropdownMenuExample: View {

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            CNDropdownMenu("Workspace actions") {
                Button("Rename") {}
                Divider()
                Button("Delete", role: .destructive) {}
            }

        }
    }
}
```

## Native behavior

Compose native content in the builder. Pass classes: to replace or extend the default recipe. Bindings and application actions remain caller-owned. Layout, semantics, and system control behavior remain native.

## Style and compose

Visual recipes use `.tw` classes. Append `classes:` to override a recipe locally, or change the recipe in `TWGlobalRules.named` for your subtree. Native layout and interaction modifiers remain SwiftUI APIs. For control colors, append `cn-tint-[primary]`, another theme token, or an interpolated `Color`.

Motion recipes are editable too: `control-motion` uses a 150 ms smooth response, `disclosure-motion` uses a 250 ms spring, and `feedback-motion` uses a 200 ms smooth response. Set a recipe to `animate-none` in your global rules to turn it off. Presets respect Reduce Motion. Static components stay still; native scrolling, drag tracking, and platform presentation keep their own gestures.

Action groups and card, dialog, and questionnaire footers stack when their content does not fit. The native layout preserves the same child controls during this change.

View builders preserve native child content. Native Button roles, labels, keyboard shortcuts, submit actions, and presentation modifiers still work. A string recipe does not replace a native control API or a typed chart mark.

```swift title="Menus.swift"
import SwiftUI
import SwiftCN

/// Native menu items keep their roles, shortcuts, submenus, and disabled behavior.
public struct CNDropdownMenu<Label: View, Content: View>: View {
    private let classes: TWClasses
    private let label: Label
    private let content: () -> Content
    public init(_ classes: TWClasses = "", @ViewBuilder content: @escaping () -> Content, @ViewBuilder label: () -> Label) {
        self.classes = classes; self.content = content; self.label = label()
    }
    public init(_ title: LocalizedStringKey, classes: TWClasses = "", @ViewBuilder content: @escaping () -> Content) where Label == Text {
        self.init(classes, content: content) { Text(title) }
    }
    public var body: some View { Menu(content: content, label: { label }).tw(cn("menu", classes)) }
}

/// Compose native links, navigation links, and submenu controls without owning the application's navigation path.
public struct CNNavigationMenu<Content: View>: View {
    private let classes: TWClasses
    private let content: Content
    public init(_ classes: TWClasses = "", @ViewBuilder content: () -> Content) { self.classes = classes; self.content = content() }
    public var body: some View {
        ScrollView(.horizontal) { HStack(spacing: 12) { content } }.tw(cn("menu", classes))
    }
}
/// In-window menu strip. For application menu-bar commands, use SwiftUI Scene.commands instead.
public struct CNMenubar<Content: View>: View {
    private let classes: TWClasses
    private let content: Content
    public init(_ classes: TWClasses = "", @ViewBuilder content: () -> Content) { self.classes = classes; self.content = content() }
    public var body: some View { HStack(spacing: 8) { content }.tw(cn("menu p-1 bg-surface border rounded-md", classes)) }
}

/// Attach native contextual actions to any content, including rows and images.
public struct CNContextMenu<Content: View, MenuItems: View>: View {
    private let classes: TWClasses
    private let content: Content
    private let menuItems: () -> MenuItems
    public init(_ classes: TWClasses = "", @ViewBuilder menuItems: @escaping () -> MenuItems, @ViewBuilder content: () -> Content) {
        self.classes = classes; self.menuItems = menuItems; self.content = content()
    }
    public var body: some View { content.tw(classes).contextMenu(menuItems: menuItems) }
}
```
